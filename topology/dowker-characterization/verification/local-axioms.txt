def Dowker.CountablyParacompact.{u} : (X : Type u) → [TopologicalSpace X] → Prop :=
fun X [TopologicalSpace X] =>
  ∀ (U : Set (Set X)),
    U.Countable →
      (∀ s ∈ U, IsOpen s) →
        ⋃₀ U = Set.univ →
          ∃ V, (∀ s ∈ V, IsOpen s) ∧ ⋃₀ V = Set.univ ∧ (LocallyFinite fun s => ↑s) ∧ ∀ s ∈ V, ∃ t ∈ U, s ⊆ t
def Dowker.StrictInsertion.{u} : (X : Type u) → [TopologicalSpace X] → Prop :=
fun X [TopologicalSpace X] =>
  ∀ (h g : X → ℝ),
    UpperSemicontinuous h →
      LowerSemicontinuous g → (∀ (x : X), h x < g x) → ∃ f, Continuous f ∧ ∀ (x : X), h x < f x ∧ f x < g x
