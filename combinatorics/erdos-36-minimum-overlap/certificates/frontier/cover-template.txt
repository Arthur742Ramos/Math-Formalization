    ∃ r ∈ positiveRows, (r.lo : ℝ) ≤ m ∧ m ≤ (r.hi : ℝ) ∧ (height : ℝ) < r.eval m := by
  exact exists_separating_row positiveRows_chain positiveRows_valid m (by simpa using hlo) (by simpa using hhi)
